            'autoconsumo': {
                name: 'Energía Solar',
                type: 'data',
                useClustering: false,
                color: '#f59e0b',
                source: 'Ministerio para la Transición Ecológica',
                _markers: [],
                _isActive: false,
                contextBounds: [[35.95, -6.5], [37.0, -5.1]],
                fetchData: async () => {
                    try {
                        const res = await fetch('/api/autoconsumo');
                        if (!res.ok) return [];
                        const json = await res.json();
                        return json.municipalities || [];
                    } catch (e) {
                        return [];
                    }
                },
                renderMarker: (city, map) => {
                    // Radius based on MW (min 10, max 40)
                    const radius = Math.min(Math.max(10, city.mw * 1.5), 50);
                    return window.L.circleMarker([city.lat, city.lon], {
                        radius: radius,
                        fillColor: '#f59e0b',
                        color: '#b45309',
                        weight: 2,
                        opacity: 0.8,
                        fillOpacity: 0.5
                    });
                },
                onMarkerClick: (city) => {
                    const html = `
                        <div style="font-family:system-ui,sans-serif;">
                            <div style="font-size:16px;font-weight:700;color:var(--text-primary);margin-bottom:8px;">
                                ${city.name}
                            </div>
                            <div style="display:flex;align-items:center;gap:12px;background:var(--chat-bg);padding:12px;border-radius:8px;border:1px solid var(--border-color);">
                                <div style="font-size:32px;">☀️</div>
                                <div>
                                    <div style="font-size:12px;color:var(--text-secondary);text-transform:uppercase;font-weight:600;">Potencia Solar</div>
                                    <div style="font-size:24px;font-weight:800;color:#f59e0b;">${city.mw} <span style="font-size:14px;color:var(--text-primary);">MW</span></div>
                                </div>
                            </div>
                            <div style="margin-top:12px;font-size:11px;color:var(--text-secondary);text-align:right;">
                                <i>Registro MITECO</i>
                            </div>
                        </div>
                    `;
                    return {
                        label: '🔆 ENERGÍA SOLAR',
                        title: city.name,
                        desc: html,
                        showActions: false,
                        recenter: [city.lat, city.lon]
                    };
                }
            },
